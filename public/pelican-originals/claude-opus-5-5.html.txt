  }
  const w = (x, n, p) => Math.sin((2 * Math.PI * n * x) / TILE + p);

  function repeatTile(group, markup) {

  `);

  let tufts = '';
  [30, 150, 265, 420, 540, 700].forEach((x) => {
    tufts += `<path class="tuft" d="M${x},493 q1,-9 3,-12 M${x + 5},493 q0,-11 1,-15 M${x + 10},493 q1,-8 4,-10"/>`;
  });
  [95, 340, 615].forEach((x) => {
    tufts += `<path class="tuft" d="M${x},494 L${x},484"/>` +
      `<circle class="petal" cx="${x}" cy="482" r="5"/><circle class="pistil" cx="${x}" cy="482" r="2"/>`;
  });
  repeatTile($('tufts'), tufts);

  // stars: deterministic scatter
  let seed = 7;
  const rand = () => ((seed = (seed * 16807) % 2147483647) / 2147483647);
  let stars = '';
  for (let i = 0; i < 1000; i++) {
    const x = -2200 + rand() * 5000, y = -1000 + rand() * 1260, r = 0.6 + rand() * 1.2;
    stars += `<circle cx="${x.toFixed(0)}" cy="${y.toFixed(0)}" r="${r.toFixed(2)}" opacity="${(0.5 + rand() * 0.5).toFixed(2)}"/>`;

  // spokes
  function spokes(cx, cy) {
    let d = '';
    for (let i = 0; i < 6; i++) {
      const a = (i * Math.PI) / 6, dx = Math.cos(a) * 45, dy = Math.sin(a) * 45;
      d += `M${(cx - dx).toFixed(1)},${(cy - dy).toFixed(1)} L${(cx + dx).toFixed(1)},${(cy + dy).toFixed(1)} `;
    }
    return d;
  }
  $('spokesR').setAttribute('d', spokes(300, 364));
  $('spokesF').setAttribute('d', spokes(495, 364));

  /* ---------- viewport fitting ---------- */
  let vb = { x: 0, y: 0, w: 800, h: 500 };
  function fit() {
    const r = stage.getBoundingClientRect();
    const a = r.width / Math.max(1, r.height);
    let W = 500 * a, H = 500;
    if (W < 370) { W = 370; H = W / a; }
    vb = { x: 400 - W / 2, y: 500 - H, w: W, h: H };
    svg.setAttribute('viewBox', `${vb.x.toFixed(1)} ${vb.y.toFixed(1)} ${W.toFixed(1)} ${H.toFixed(1)}`);
    $('skyGrad').setAttribute('y1', vb.y.toFixed(1));
    const cy = vb.y + Math.max(70, Math.min(100, H * 0.18));
    $('celestial').setAttribute('transform', `translate(${(vb.x + W * 0.8).toFixed(1)},${cy.toFixed(1)})`);
    dirty = true;
  }

  /* ---------- rider animation ---------- */
  const BB = { x: 385, y: 370 };
  const CRANK = 20, L1 = 58, L2 = 56, WHEEL_R = 56, BASE_SPEED = 230;
  const els = {
    wheelR: $('wheelR'), wheelF: $('wheelF'), chain: $('chain'),
    crankN: $('crankNear'), crankF: $('crankFar'), pedalN: $('pedalNear'), pedalF: $('pedalFar'),
    leg: $('leg'), legInk: $('legInk'), paw: $('paw'),
    legF: $('legFar'), legFInk: $('legFarInk'), pawF: $('pawFar'),
    arm: $('arm'), armInk: $('armInk'), pawArm: $('pawArm'),
    cat: $('catMain'), haunch: $('haunch'), tail: $('tail'),
    scarfA: $('scarfTailA'), scarfB: $('scarfTailB'),
    eyeN: $('eyeNear'), eyeF: $('eyeFar'),
    hillsFar: $('hillsFar'), hillsNear: $('hillsNear'), trees: $('trees'),
    clouds: $('clouds'), tufts: $('tufts'), dash: $('dash'),
  };

  function ik(hx, hy, fx, fy) {
    let dx = fx - hx, dy = fy - hy, d = Math.hypot(dx, dy);
    const maxd = L1 + L2 - 0.5;
    if (d > maxd) { fx = hx + (dx / d) * maxd; fy = hy + (dy / d) * maxd; dx = fx - hx; dy = fy - hy; d = maxd; }
    const a = (L1 * L1 - L2 * L2 + d * d) / (2 * d);
    const h = Math.sqrt(Math.max(0, L1 * L1 - a * a));
    const ux = dx / d, uy = dy / d;
    return { kx: hx + a * ux + h * uy, ky: hy + a * uy - h * ux, fx, fy };
  }
  const f1 = (n) => n.toFixed(1);

  function setLeg(pathInk, path, paw, hx, hy, ang) {
    const px = BB.x + CRANK * Math.cos(ang), py = BB.y + CRANK * Math.sin(ang);
    const j = ik(hx, hy, px, py - 6);
    const d = `M${f1(hx)},${f1(hy)} L${f1(j.kx)},${f1(j.ky)} L${f1(j.fx)},${f1(j.fy)}`;
    pathInk.setAttribute('d', d);
    path.setAttribute('d', d);
    paw.setAttribute('cx', f1(j.fx + 3));
    paw.setAttribute('cy', f1(j.fy));
    return { px, py };
  }

  function ribbon(x0, y0, len, rise, width, amp, phase) {
    const N = 14, top = [], bot = [];
    for (let i = 0; i <= N; i++) {
      const s = i / N;
      const x = x0 - len * s;
      const y = y0 - rise * s + amp * s * Math.sin(phase - s * 5.5);
      const hw = (width / 2) * (1 - 0.3 * s);
      top.push(`${f1(x)},${f1(y - hw)}`);
      bot.unshift(`${f1(x)},${f1(y + hw)}`);
    }
    // small notch at the loose end
    const s = 1, xe = x0 - len - 5, ye = y0 - rise + amp * Math.sin(phase - 5.5);
    return `M${top.join(' L')} L${f1(xe)},${f1(ye - 1)} L${f1(x0 - len + 2)},${f1(ye)} L${f1(xe)},${f1(ye + 1)} L${bot.join(' L')} Z`;
  }

  let dist = 0, crank = -Math.PI / 3, time = 0, speedMul = 1, blinkAt = 2.5, blinkUntil = 0;
  let dirty = true;

  function render() {
    const wheelDeg = ((dist / WHEEL_R) * 180) / Math.PI;
    els.wheelR.setAttribute('transform', `rotate(${f1(wheelDeg % 360)} 300 364)`);
    els.wheelF.setAttribute('transform', `rotate(${f1(wheelDeg % 360)} 495 364)`);
    els.chain.style.strokeDashoffset = f1(-(crank * 15) % 52);

    const bob = 1.3 * Math.sin(2 * crank);
    els.cat.setAttribute('transform', `translate(0,${f1(bob)})`);
    els.haunch.setAttribute('transform', `translate(0,${f1(bob)})`);

    const pn = setLeg(els.legInk, els.leg, els.paw, 372, 276 + bob, crank);
    const pf = setLeg(els.legFInk, els.legF, els.pawF, 380, 272 + bob, crank + Math.PI);
    els.crankN.setAttribute('x2', f1(pn.px)); els.crankN.setAttribute('y2', f1(pn.py));
    els.crankF.setAttribute('x2', f1(pf.px)); els.crankF.setAttribute('y2', f1(pf.py));
    els.pedalN.setAttribute('x', f1(pn.px - 11)); els.pedalN.setAttribute('y', f1(pn.py - 3));
    els.pedalF.setAttribute('x', f1(pf.px - 11)); els.pedalF.setAttribute('y', f1(pf.py - 3));

    const armD = `M440,${f1(252 + bob)} Q463,${f1(270 + bob * 0.5)} 487,262`;
    els.arm.setAttribute('d', armD);
    els.armInk.setAttribute('d', armD);

    els.tail.setAttribute('transform', `rotate(${f1(7 * Math.sin(time * 2.1))} 342 266)`);

    const flutter = 3 + 3 * speedMul;
    const ph = time * (6 + 4 * speedMul);
    els.scarfA.setAttribute('d', ribbon(437, 244, 62 + 12 * speedMul, 26, 11, flutter, ph));
    els.scarfB.setAttribute('d', ribbon(437, 249, 48 + 10 * speedMul, 14, 9, flutter * 0.8, ph + 1.7));

    const closed = time < blinkUntil;
    els.eyeN.setAttribute('ry', closed ? 1.2 : 7.5);
    els.eyeF.setAttribute('ry', closed ? 1.2 : 7);

    const par = (g, f) => g.setAttribute('transform', `translate(${f1(-((dist * f) % TILE))},0)`);
    par(els.clouds, 0.06);
    par(els.hillsFar, 0.15);
    par(els.hillsNear, 0.35);
    par(els.trees, 0.8);
    par(els.tufts, 1.15);
    els.dash.style.strokeDashoffset = f1(dist % 84);
  }

  const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  let running = !reduceMotion;
  let last = performance.now();

  function loop(now) {
    const dt = Math.min(0.05, (now - last) / 1000);
    last = now;
    if (running) {
      time += dt;
      const v = BASE_SPEED * speedMul;
      dist += v * dt;
      crank += (v / WHEEL_R) * 0.85 * dt;
      if (time > blinkAt) { blinkUntil = time + 0.13; blinkAt = time + 2.5 + Math.random() * 3; }
      dirty = true;
    }
    if (dirty) { render(); dirty = false; }
    requestAnimationFrame(loop);
  }

  /* ---------- controls ---------- */
  const playBtn = $('playBtn');
  function syncPlay() { playBtn.textContent = running ? '暂停' : '出发'; }
  playBtn.addEventListener('click', () => { running = !running; syncPlay(); });
  syncPlay();

  const speed = $('speed'), speedOut = $('speedOut');
  speed.addEventListener('input', () => {
    speedMul = parseFloat(speed.value);
    speedOut.textContent = speedMul.toFixed(1) + '×';
    dirty = true;
  });

  let audio = null;
  function ding() {
    try {
      audio = audio || new (window.AudioContext || window.webkitAudioContext)();
      if (audio.state === 'suspended') audio.resume();
      const t0 = audio.currentTime;
      [0, 0.17].forEach((delay) => {
        [[2350, 0.22], [3520, 0.08], [5100, 0.03]].forEach(([freq, vol]) => {
          const o = audio.createOscillator(), g = audio.createGain();
          o.type = 'sine';
          o.frequency.value = freq;
          g.gain.setValueAtTime(0.0001, t0 + delay);
          g.gain.exponentialRampToValueAtTime(vol, t0 + delay + 0.005);
          g.gain.exponentialRampToValueAtTime(0.0001, t0 + delay + 0.7);
          o.connect(g).connect(audio.destination);
          o.start(t0 + delay);
          o.stop(t0 + delay + 0.75);
        });
      });
    } catch (e) { /* sound is optional */ }
  }

  const bell = $('bell'), bubble = $('bubble');
  let bubbleTimer = null;
  function ring() {
    ding();
    bell.classList.remove('ring');
    void bell.getBoundingClientRect();
    bell.classList.add('ring');
    bubble.classList.add('show');
    clearTimeout(bubbleTimer);
    bubbleTimer = setTimeout(() => bubble.classList.remove('show'), 1300);
  }
  $('bellBtn').addEventListener('click', ring);
  $('rider').addEventListener('click', ring);

  const themeBtn = $('themeBtn');
  const darkQuery = window.matchMedia('(prefers-color-scheme: dark)');
  const isDark = () => {
    const t = document.documentElement.getAttribute('data-theme');
    return t ? t === 'dark' : darkQuery.matches;
  };
  function syncTheme() { themeBtn.textContent = isDark() ? '换成白天' : '换成夜晚'; }
  themeBtn.addEventListener('click', () => {
    document.documentElement.setAttribute('data-theme', isDark() ? 'light' : 'dark');
    syncTheme();
  });
  try { darkQuery.addEventListener('change', syncTheme); } catch (e) { /* older browsers */ }
  syncTheme();

  if ('ResizeObserver' in window) new ResizeObserver(fit).observe(stage);
  window.addEventListener('resize', fit);
  fit();
  requestAnimationFrame(loop);
})();
