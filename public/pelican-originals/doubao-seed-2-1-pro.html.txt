<script>
(function () {
  var $ = function (id) { return document.getElementById(id); };
  var stage = $('stage'), toggleBtn = $('toggleBtn'), bellBtn = $('bellBtn');
  var spokesRear = $('spokesRear'), spokesFront = $('spokesFront');
  var crankG = $('crankG'), pedalAG = $('pedalAG'), pedalBG = $('pedalBG');
  var legFarOut = $('legFarOut'), legFarIn = $('legFarIn');
  var legNearOut = $('legNearOut'), legNearIn = $('legNearIn');
  var pawA = $('pawA'), pawB = $('pawB');
  var scenery = $('scenery'), hillsFar = $('hillsFar'), hillsNear = $('hillsNear');
  var catRider = $('catRider'), bellRinger = $('bellRinger'), dingBubble = $('dingBubble');

  var ICON_PAUSE = '<svg width="15" height="15" viewBox="0 0 24 24" fill="currentColor"><rect x="6" y="4" width="4" height="16" rx="1.5"/><rect x="14" y="4" width="4" height="16" rx="1.5"/></svg><span>暂停</span>';
  var ICON_PLAY = '<svg width="15" height="15" viewBox="0 0 24 24" fill="currentColor"><path d="M7 4.8v14.4c0 1 1.1 1.6 2 1.1l11-7.2c.8-.5.8-1.7 0-2.2L9 3.7c-.9-.5-2 .1-2 1.1z"/></svg><span>继续</span>';

  // ---- 骑行状态 ----
  var CRANK_RATE = 2.8;          // 曲柄角速度 rad/s（正常档）
  var GEAR_RATIO = 2.2;          // 车轮 / 曲柄传动比
  var WHEEL_R = 70;              // 车轮半径 px
  var CRANK_R = 22;              // 曲柄长度 px
  var BB = { x: 490, y: 390 };   // 中轴
  var HIP_FAR = { x: 428, y: 262 };
  var HIP_NEAR = { x: 448, y: 266 };
  var THIGH = 84, SHIN = 82;     // 大腿 / 小腿长度

  var crank = 0, dist = 0, running = true, last = performance.now();

  // 两骨反向运动学：给定髋点、踝点与两段腿长，求朝上的膝盖
  function knee(hx, hy, px, py, l1, l2) {
    var dx = px - hx, dy = py - hy;
    var raw = Math.hypot(dx, dy) || 1;
    var d = Math.min(Math.max(raw, Math.abs(l1 - l2) + .5), l1 + l2 - .5);
    var ux = dx / raw, uy = dy / raw;
    var along = (l1 * l1 - l2 * l2 + d * d) / (2 * d);
    var h = Math.sqrt(Math.max(0, l1 * l1 - along * along));
    var mx = hx + ux * along, my = hy + uy * along;
    var k1 = { x: mx - uy * h, y: my + ux * h };
    var k2 = { x: mx + uy * h, y: my - ux * h };
    return k1.y <= k2.y ? k1 : k2;
  }

  function legPath(hip, paw, l1, l2) {
    var k = knee(hip.x, hip.y, paw.x, paw.y, l1, l2);
    return 'M' + hip.x + ',' + hip.y + ' L' + k.x.toFixed(1) + ',' + k.y.toFixed(1) + ' L' + paw.x.toFixed(1) + ',' + paw.y.toFixed(1);
  }

  function update() {
    var deg = crank * 180 / Math.PI;
    var wheelDeg = deg * GEAR_RATIO;

    // 车轮辐条
    spokesRear.setAttribute('transform', 'rotate(' + wheelDeg + ')');
    spokesFront.setAttribute('transform', 'rotate(' + wheelDeg + ')');

    // 曲柄与踏板（踏板始终保持水平）
    crankG.setAttribute('transform', 'rotate(' + deg + ')');
    pedalAG.setAttribute('transform', 'translate(0,22) rotate(' + (-deg) + ')');
    pedalBG.setAttribute('transform', 'translate(0,-22) rotate(' + (-deg) + ')');

    // 两个踏板的世界坐标
    var p1 = { x: BB.x + CRANK_R * Math.sin(crank), y: BB.y + CRANK_R * Math.cos(crank) };
    var p2 = { x: BB.x - CRANK_R * Math.sin(crank), y: BB.y - CRANK_R * Math.cos(crank) };

    // 车身随踩踏节奏轻微起伏
    var bounce = (1 - Math.cos(2 * crank)) / 2 * 2;
    catRider.setAttribute('transform', 'translate(0,' + (-bounce).toFixed(2) + ')');

    // 猫腿（髋点随身体起伏）
    var hipFar = { x: HIP_FAR.x, y: HIP_FAR.y - bounce };
    var hipNear = { x: HIP_NEAR.x, y: HIP_NEAR.y - bounce };
    var dFar = legPath(hipFar, p1, THIGH, SHIN);
    var dNear = legPath(hipNear, p2, THIGH, SHIN);
    legFarOut.setAttribute('d', dFar); legFarIn.setAttribute('d', dFar);
    legNearOut.setAttribute('d', dNear); legNearIn.setAttribute('d', dNear);

    // 脚爪贴合踏板
    pawA.setAttribute('cx', p1.x.toFixed(1)); pawA.setAttribute('cy', p1.y.toFixed(1));
    pawB.setAttribute('cx', p2.x.toFixed(1)); pawB.setAttribute('cy', p2.y.toFixed(1));

    // 背景滚动（地面与车轮同步，远山视差）
    scenery.setAttribute('transform', 'translate(' + (-(dist % 960)).toFixed(1) + ',0)');
    hillsFar.setAttribute('transform', 'translate(' + (-(dist * .15 % 960)).toFixed(1) + ',0)');
    hillsNear.setAttribute('transform', 'translate(' + (-(dist * .32 % 960)).toFixed(1) + ',0)');
  }

  function frame(now) {
    var dt = Math.min((now - last) / 1000, .05);
    last = now;
    if (running) {
      crank += dt * CRANK_RATE;
      dist += dt * CRANK_RATE * GEAR_RATIO * WHEEL_R;
      update();
    }
    requestAnimationFrame(frame);
  }

  // ---- 暂停 / 继续 ----
  function setRunning(v) {
    running = v;
    stage.classList.toggle('paused', !v);
    toggleBtn.innerHTML = v ? ICON_PAUSE : ICON_PLAY;
  }
  toggleBtn.addEventListener('click', function () { setRunning(!running); });

  // ---- 速度档位 ----
  var segBtns = document.querySelectorAll('.seg button');
  segBtns.forEach(function (btn) {
    btn.addEventListener('click', function () {
      CRANK_RATE = parseFloat(btn.dataset.rate);
      segBtns.forEach(function (b) { b.classList.remove('active'); });
      btn.classList.add('active');
    });
  });

  // ---- 车铃：摆动 + 气泡 + 铃声 ----
  var actx = null;
  function playDing() {
    try {
      actx = actx || new (window.AudioContext || window.webkitAudioContext)();
      if (actx.state === 'suspended') { actx.resume(); }
      var t0 = actx.currentTime;
      [[1567.98, .16, 0], [2093.0, .10, .03]].forEach(function (n) {
        var o = actx.createOscillator(), g = actx.createGain();
        o.type = 'sine';
        o.frequency.value = n[0];
        g.gain.setValueAtTime(.0001, t0 + n[2]);
        g.gain.exponentialRampToValueAtTime(n[1], t0 + n[2] + .012);
        g.gain.exponentialRampToValueAtTime(.0001, t0 + n[2] + .55);
        o.connect(g); g.connect(actx.destination);
        o.start(t0 + n[2]); o.stop(t0 + n[2] + .6);
      });
    } catch (e) { /* 音频不可用时静默 */ }
  }
  function restartClass(el, cls) {
    el.classList.remove(cls);
    void el.getBoundingClientRect();
    el.classList.add(cls);
  }
  bellBtn.addEventListener('click', function () {
    restartClass(bellRinger, 'ringing');
    restartClass(dingBubble, 'show');
    playDing();
  });

  // ---- 初始化 ----
  var reduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  setRunning(!reduced);
  update();
  requestAnimationFrame(frame);
})();
</script>
