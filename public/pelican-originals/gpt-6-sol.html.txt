<script>
  (() => {
    const scene = document.getElementById('scene');
    const svg = scene.querySelector('svg');
    const speedInput = document.getElementById('speed');
    const speedValue = document.getElementById('speedValue');
    const toggle = document.getElementById('toggle');
    const reduced = matchMedia('(prefers-reduced-motion: reduce)');
    const parts = Object.fromEntries(['farLeg','farPaw','nearLeg','nearLegOutline','nearPaw','rearSpokes','frontSpokes','cranks'].map(id => [id, svg.getElementById(id)]));
    const C = { x: 485, y: 438 }, R = 53;
    let angle = -.65, last = 0, paused = reduced.matches, speed = 1;

    // Two-bone inverse kinematics keeps each paw on its pedal throughout a revolution.
    function leg(hip, foot, upper, lower) {
      const dx = foot.x - hip.x, dy = foot.y - hip.y;
      const distance = Math.hypot(dx, dy);
      const d = Math.min(Math.max(distance, Math.abs(upper - lower) + .01), upper + lower - .01);
      const ux = dx / distance, uy = dy / distance;
      const a = (upper * upper - lower * lower + d * d) / (2 * d);
      const h = Math.sqrt(Math.max(0, upper * upper - a * a));
      const knee = { x: hip.x + ux * a + uy * h, y: hip.y + uy * a - ux * h };
      return `M${hip.x.toFixed(1)} ${hip.y.toFixed(1)} L${knee.x.toFixed(1)} ${knee.y.toFixed(1)} L${foot.x.toFixed(1)} ${foot.y.toFixed(1)}`;
    }
    function position() {
      const near = { x: C.x + R * Math.cos(angle), y: C.y + R * Math.sin(angle) };
      const far = { x: C.x - R * Math.cos(angle), y: C.y - R * Math.sin(angle) };
      const n = leg({x: 449, y: 284}, near, 112, 112);
      parts.nearLeg.setAttribute('d', n);
      parts.nearLegOutline.setAttribute('d', n);
      parts.farLeg.setAttribute('d', leg({x: 425, y: 286}, far, 113, 113));
      for (const [paw, point] of [[parts.nearPaw, near], [parts.farPaw, far]]) {
        paw.setAttribute('cx', point.x.toFixed(1));
        paw.setAttribute('cy', point.y.toFixed(1));
        paw.setAttribute('transform', `rotate(-15 ${point.x.toFixed(1)} ${point.y.toFixed(1)})`);
      }
      const deg = angle * 180 / Math.PI;
      parts.cranks.setAttribute('transform', `rotate(${deg} ${C.x} ${C.y})`);
      parts.rearSpokes.setAttribute('transform', `rotate(${deg * 1.9} 300 445)`);
      parts.frontSpokes.setAttribute('transform', `rotate(${deg * 1.9} 690 445)`);
    }
    function frame(time) {
      if (!last) last = time;
      const dt = Math.min((time - last) / 1000, .05);
      last = time;
      if (!paused) { angle += dt * 2.7 * speed; position(); }
      requestAnimationFrame(frame);
    }
    function refresh() {
      scene.classList.toggle('paused', paused);
      toggle.textContent = paused ? '播放 ▶' : '暂停 Ⅱ';
      toggle.setAttribute('aria-pressed', String(paused));
    }
    speedInput.addEventListener('input', () => {
      speed = Number(speedInput.value);
      speedValue.value = `${speed.toFixed(1)}×`;
      scene.style.setProperty('--speed', speed);
      scene.querySelectorAll('.roadMarks, .scarf, .tail, .cat-body, .wind').forEach(el => {
        const base = el.classList.contains('roadMarks') ? .95 : el.classList.contains('scarf') ? .85 : el.classList.contains('tail') ? .75 : el.classList.contains('cat-body') ? .36 : 1.3;
        el.style.animationDuration = `${base / speed}s`;
      });
    });
    toggle.addEventListener('click', () => { paused = !paused; refresh(); });
    reduced.addEventListener('change', e => { paused = e.matches; refresh(); });
    position(); refresh(); requestAnimationFrame(frame);
  })();
</script>
