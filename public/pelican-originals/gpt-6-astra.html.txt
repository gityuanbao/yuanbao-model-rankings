  // All artwork is inline SVG; the page works completely offline.
  (() => {
    'use strict';
    const $ = id => document.getElementById(id);
    const ns = 'http://www.w3.org/2000/svg';
    const node = (tag, attrs, parent) => {
      const el = document.createElementNS(ns, tag);
      Object.entries(attrs).forEach(([key, value]) => el.setAttribute(key, value));
      parent.appendChild(el);
      return el;
    };
    const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
    let paused = reducedMotion.matches, time = 0, last = null, speedIndex = 1;
    let bellUntil = 0, audioContext = null, messageTimer;
    const speeds = [{value:.5, text:'慢慢 · 0.5×', label:'慢慢', display:.5}, {value:1, text:'悠闲 · 1×', label:'悠闲', display:1}, {value:1.7, text:'追风 · 1.7×', label:'追风', display:1.7}];
    const spokes = [$('backSpokes'), $('frontSpokes')];
    [453, 752].forEach((cx, wheel) => {
      for(let i = 0; i < 12; i++) {
        const angle = i * Math.PI / 6;
        node('line', {x1:cx, y1:505, x2:cx + 72 * Math.cos(angle), y2:505 + 72 * Math.sin(angle)}, spokes[wheel]);
      }
      node('rect', {x:cx+47, y:501, width:14, height:7, rx:2, fill:'#e5bc73', stroke:'none'}, spokes[wheel]);
    });
    const background = [], foreground = [], road = [];
    for(let i = 0; i < 7; i++) {
      const x = i * 260 - 200, scale = [.72, .48, .68, .52, .82, .55, .7][i];
      const tree = node('use', {href:'#tree'}, $('scenery'));
      background.push({el:tree, x, y:488 + i%2*9, scale});
      const bush = node('use', {href:'#bush'}, $('scenery'));
      background.push({el:bush, x:x+122, y:526, scale:.7});
    }
    for(let i = 0; i < 19; i++) {
      const isFlower = i % 3 === 0;
      const el = node('use', {href:isFlower ? '#daisy' : '#grass'}, $('foreground'));
      foreground.push({el, x:i*86-120, y:640+(i%3)*9, scale:isFlower ? .62+(i%2)*.2 : .55+(i%4)*.13});
    }
    for(let i = 0; i < 12; i++) {
      const el = node('path', {d:'M0 0h'+(12+i%3*13)}, $('roadMarks'));
      road.push({el, x:i*125, y:600+(i%3)*8});
    }
    const wrap = (x, travel, width, start) => ((x-travel-start)%width+width)%width+start;
    const set = (id, attrs) => Object.entries(attrs).forEach(([key, val]) => $(id).setAttribute(key, val));
    // Solve the knee position from a fixed hip and moving pedal endpoint.
    function leg(hipX, hipY, footX, footY, prefix) {
      const dx=footX-hipX, dy=footY-hipY, d=Math.hypot(dx,dy);
      const length=84, bend=Math.sqrt(Math.max(0,length*length-d*d/4));
      const kneeX=(hipX+footX)/2 + dy/d*bend;
      const kneeY=(hipY+footY)/2 - dx/d*bend;
      const path=`M${hipX} ${hipY}L${kneeX} ${kneeY}L${footX} ${footY-5}`;
      set(prefix, {d:path});
      set(prefix+'Outline', {d:path});
    }
    function draw() {
      const travel=time*111, angle=time*2.15, degrees=travel/82*180/Math.PI;
      set('backSpokes', {transform:`rotate(${degrees} 453 505)`});
      set('frontSpokes', {transform:`rotate(${degrees} 752 505)`});
      const px=597+32*Math.cos(angle), py=505+32*Math.sin(angle);
      const fx=597-32*Math.cos(angle), fy=505-32*Math.sin(angle);
      set('crank', {d:`M${fx} ${fy}L${px} ${py}`});
      leg(578, 389, fx, fy, 'legFar');
      leg(580, 386, px, py, 'legNear');
      set('pawNear', {transform:`translate(${px} ${py})`});
      set('pedalNear', {transform:`translate(${px} ${py})`});
      set('pedalFar', {transform:`translate(${fx} ${fy})`});
      const sway=Math.sin(time*2.15), flutter=Math.sin(time*4.2);
      set('rider', {transform:`translate(0 ${Math.sin(time*4.3)*.65})`});
      const tail=`M557 364C504 ${365+sway*4} 512 ${308+sway*8} 475 ${312+sway*9}S446 ${362+sway*5} 472 ${351+sway*4}`;
      set('tail', {d:tail}); set('tailOutline', {d:tail});
      set('scarf', {d:`M581 292C550 ${280+flutter*5} 528 ${300+flutter*7} 504 ${283+flutter*10}L509 ${305+flutter*10}C542 ${319+flutter*6} 549 ${295+flutter*4} 579 310Z`});
      set('catHead', {transform:`rotate(${sway*.8} 611 286)`});
      const blink=time%5.4 > 5.19 ? .13 : 1;
      set('catEyes', {transform:`translate(0 ${237*(1-blink)}) scale(1 ${blink})`});
      set('bouquet', {transform:`rotate(${Math.sin(time*2.8)*1.5} 790 377)`});
      set('clouds', {transform:`translate(${-time*3%150} 0)`});
      set('hillsFar', {transform:`translate(${-travel*.065%500} 0)`});
      set('hillsNear', {transform:`translate(${-travel*.13%540} 0)`});
      for(const item of background) item.el.setAttribute('transform',`translate(${wrap(item.x,travel*.34,1820,-260)} ${item.y}) scale(${item.scale})`);
      for(const item of foreground) item.el.setAttribute('transform',`translate(${wrap(item.x,travel,1634,-120)} ${item.y}) scale(${item.scale})`);
      for(const item of road) item.el.setAttribute('transform',`translate(${wrap(item.x,travel,1500,-100)} ${item.y})`);
      set('butterfly', {transform:`translate(${922+Math.sin(time*.7)*31} ${329+Math.sin(time*1.8)*18}) rotate(${Math.sin(time*2)*9}) scale(${.65+Math.abs(Math.sin(time*7))*.3} .82)`});
    }
    function updateControls() {
      document.body.classList.toggle('paused',paused);
      $('playToggle').setAttribute('aria-pressed',String(paused));
      $('playToggle').setAttribute('aria-label',paused?'继续动画':'暂停动画');
      $('playLabel').textContent=paused?'继续兜风':'歇一小会';
      $('playIcon').innerHTML=paused?'<path d="m6 3 11 7-11 7Z"/>':'<path d="M5 4h3v12H5zm7 0h3v12h-3z"/>';
      $('rideStatus').textContent=paused?'在这里，发一小会儿呆':'正在收集路上的小美好';
    }
    $('playToggle').addEventListener('click',()=>{paused=!paused;updateControls();});
    $('speedToggle').addEventListener('click',()=>{
      speedIndex=(speedIndex+1)%speeds.length;
      const speed=speeds[speedIndex];
      $('speedLabel').textContent=speed.text;
      $('speedToggle').setAttribute('aria-label',`调整骑行速度，当前${speed.label}，${speed.display}倍`);
    });
    $('bellButton').addEventListener('click',async()=>{
      bellUntil=performance.now()+950;
      $('bellMessage').textContent='叮铃～ 你好呀！';
      $('bellMessage').classList.add('visible');
      clearTimeout(messageTimer);
      messageTimer=setTimeout(()=>$('bellMessage').classList.remove('visible'),1700);
      try {
        const Audio=window.AudioContext||window.webkitAudioContext;
        if(!Audio) return;
        audioContext ||= new Audio();
        if(audioContext.state==='suspended') await audioContext.resume();
        const start=audioContext.currentTime;
        [0,.13].forEach((delay,i)=>{
          [1,2.76].forEach((harmonic)=>{
            const oscillator=audioContext.createOscillator(), gain=audioContext.createGain();
            oscillator.type='sine';oscillator.frequency.value=(i?1568:2093)*harmonic;
            gain.gain.setValueAtTime(0,start+delay);
            gain.gain.linearRampToValueAtTime(harmonic===1?.09:.025,start+delay+.006);
            gain.gain.exponentialRampToValueAtTime(.0001,start+delay+.65);
            oscillator.connect(gain);gain.connect(audioContext.destination);
            oscillator.start(start+delay);oscillator.stop(start+delay+.7);
            oscillator.onended=()=>{oscillator.disconnect();gain.disconnect();};
          });
        });
      } catch { /* The visual bell still works when browser audio is unavailable. */ }
    });
    reducedMotion.addEventListener('change',event=>{paused=event.matches;updateControls();});
    document.addEventListener('visibilitychange',()=>{last=null;});
    function frame(now) {
      const delta=last===null?0:Math.min((now-last)/1000,.05);
      last=now;
      if(!paused&&!document.hidden){time+=delta*speeds[speedIndex].value;draw();}
      const ringing=now<bellUntil;
      set('bellRays',{opacity:ringing?.5+.5*Math.sin(now*.028):0});
      requestAnimationFrame(frame);
    }
    updateControls();draw();requestAnimationFrame(frame);
  })();
  </script>
