  <script>
    (() => {
      'use strict';
      const $ = id => document.getElementById(id);
      const ui = Object.fromEntries(['play','play-text','play-icon','speed','speed-value','mood','mood-text','scene-label','rear-spokes','front-spokes','crank','near-leg-outline','near-leg-fill','far-leg-outline','far-leg-fill','near-paw','far-paw','cat-body','tail','tail-white','scarf-tail','eyes','closed-eyes','wind','clouds','trees','road-marks','flowers-back','flowers-front'].map(id => [id, $(id)]));
      const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
      let paused = reducedMotion.matches, speed = 1, time = 0, last = null, frame = null;
      // Two connected bones keep the feet attached to the pedals for a full turn.
      function leg(angle, far, bounce) {
        const hip = {x:549, y:324 + bounce};
        const foot = {x:570 + Math.cos(angle)*31, y:466 + Math.sin(angle)*31};
        const dx = foot.x-hip.x, dy = foot.y-hip.y, distance = Math.hypot(dx,dy);
        const upper = 86, lower = 97;
        const along = (upper*upper-lower*lower+distance*distance)/(2*distance);
        const perpendicular = Math.sqrt(Math.max(0,upper*upper-along*along));
        const knee = {x:hip.x+dx/distance*along+dy/distance*perpendicular, y:hip.y+dy/distance*along-dx/distance*perpendicular};
        const path = `M${hip.x.toFixed(2)} ${hip.y.toFixed(2)}L${knee.x.toFixed(2)} ${knee.y.toFixed(2)}L${foot.x.toFixed(2)} ${foot.y.toFixed(2)}`;
        const name = far ? 'far' : 'near';
        ui[name+'-leg-outline'].setAttribute('d',path);
        ui[name+'-leg-fill'].setAttribute('d',path);
        ui[name+'-paw'].setAttribute('transform',`translate(${foot.x} ${foot.y})`);
        return foot;
      }
      function draw() {
        const wheelAngle = time*2.2, pedalAngle = wheelAngle/1.7;
        const bounce = Math.sin(pedalAngle*2)*1.8;
        const degrees = wheelAngle*180/Math.PI;
        ui['rear-spokes'].setAttribute('transform',`rotate(${degrees})`);
        ui['front-spokes'].setAttribute('transform',`rotate(${degrees})`);
        const near = leg(pedalAngle,false,bounce), far = leg(pedalAngle+Math.PI,true,bounce);
        ui.crank.setAttribute('d',`M${near.x} ${near.y}L${far.x} ${far.y}`);
        ui['cat-body'].setAttribute('transform',`translate(0 ${bounce})`);
        const wag = Math.sin(time*3)*6;
        const tail = `M530 310C492 ${331+wag} 470 ${302+wag} 447 ${305+wag}C422 ${309+wag} 429 ${328+wag} 447 ${325+wag}`;
        ui.tail.setAttribute('d',tail); ui['tail-white'].setAttribute('d',tail);
        ui['scarf-tail'].setAttribute('d',`M571 253C552 257 537 ${250+wag*.4} 519 ${254+wag*.6}C505 ${257+wag*.7} 491 ${254+wag} 479 ${246+wag}L492 ${270+wag}Q532 ${276+wag*.4} 578 264Z`);
        const blink = time%4.7>4.52;
        ui.eyes.setAttribute('display',blink?'none':'inline');
        ui['closed-eyes'].setAttribute('display',blink?'inline':'none');
        ui.wind.setAttribute('opacity',String(.35+Math.sin(time*2)*.15));
        ui.wind.setAttribute('transform',`translate(${-Math.sin(time*2)*8} 0)`);
        // Repeated scenery wraps beyond the viewBox, so no visible jumps occur.
        ui.clouds.setAttribute('transform',`translate(${-((time*12)%1400)} 0)`);
        ui.trees.setAttribute('transform',`translate(${-((time*53)%1500)} 0)`);
        ui['road-marks'].setAttribute('transform',`translate(${-((time*202.4)%1450)} 0)`);
        ui['flowers-back'].setAttribute('transform',`translate(${-((time*110)%1450)} 0)`);
        ui['flowers-front'].setAttribute('transform',`translate(${-((time*215)%1500)} 0)`);
      }
      // Clone each scenery strip on both sides to make the looping panorama seamless.
      for (const [name,width] of [['clouds',1400],['trees',1500],['road-marks',1450],['flowers-back',1450],['flowers-front',1500]]) {
        const group = ui[name], originals = [...group.children];
        for (const offset of [-width,width]) {
          const copy = document.createElementNS('http://www.w3.org/2000/svg','g');
          copy.setAttribute('transform',`translate(${offset} 0)`);
          originals.forEach(child => copy.appendChild(child.cloneNode(true)));
          group.appendChild(copy);
        }
      }
      function tick(now) {
        frame = null;
        if (paused || document.hidden) { last = null; return; }
        if (last !== null) time += Math.min((now-last)/1000,.05)*speed;
        last = now; draw(); frame = requestAnimationFrame(tick);
      }
      function refreshPlayback() {
        ui.play.setAttribute('aria-pressed',String(paused));
        ui['play-text'].textContent = paused ? '继续兜风' : '歇一小会';
        ui['play-icon'].setAttribute('d',paused?'M6 3L16 10L6 17Z':'M5 4H8V16H5ZM12 4H15V16H12Z');
        if (frame !== null) cancelAnimationFrame(frame);
